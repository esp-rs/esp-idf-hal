Const equivalent of <[T]>::chunks

§Example

use konst::iter::for_each;
use konst::slice;

const CHUNKS: &[&[u8]] = &{
    let mut out = [&[] as &[u8]; 3] ;
    let fibb = &[3, 5, 8, 13, 21, 34, 55, 89];
    for_each!{(i, chunk) in slice::chunks(fibb, 3),enumerate() =>
        out[i] = chunk;
    }
    out
};

let expected: &[&[u8]] = &[&[3, 5, 8], &[13, 21, 34], &[55, 89]];

assert_eq!(CHUNKS, expected)
