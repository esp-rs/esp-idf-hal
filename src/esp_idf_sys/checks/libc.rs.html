1//! Compile-time checks that ESP-IDF bindgen types and constants match libc definitions.
2//! If any of these checks fail, it is likely you need to update `libc` pinned version
3//! to the latest one.
4//!
5//! The socket and netdb items (`sockaddr*`, `MSG_*`, `SO_*`, `IP_*`, `AI_*`, `NI_*`, ...) are only
6//! present in the bindings when the `lwip` ESP-IDF component is built, hence their checks
7//! are gated on `esp_idf_comp_lwip_enabled`.
8
9use crate as sys;
10use compile_fmt::{compile_assert, fmt};
11
12macro_rules! check_constants {
13    ($ident:ident) => {
14        const _: () = {
15            compile_assert!(
16                sys::$ident as i64 == libc::$ident as i64,
17                "libc/esp-idf-sys constant mismatch for `", stringify!($ident), "`: esp-idf=",
18                sys::$ident as i64 => fmt::<i64>(),
19                " libc=", libc::$ident as i64 => fmt::<i64>()
20            );
21        };
22    };
23}
24
25macro_rules! check_constants_manually {
26    ($name:literal, $const1:expr, $const2:expr) => {
27        const _: () = {
28            compile_assert!(
29                $const1 as i64 == $const2 as i64,
30                "libc/esp-idf-sys constant mismatch for `", $name, "`: lhs=",
31                $const1 as i64 => fmt::<i64>(),
32                " rhs=", $const2 as i64 => fmt::<i64>()
33            );
34        };
35    };
36}
37
38macro_rules! check_types {
39    ($ident:ident) => {
40        const _: () = {
41            compile_assert!(
42                std::mem::size_of::<sys::$ident>() == std::mem::size_of::<libc::$ident>(),
43                "libc/esp-idf-sys type mismatch for `", stringify!($ident), "` size: esp-idf=",
44                std::mem::size_of::<sys::$ident>() => fmt::<usize>(),
45                " libc=", std::mem::size_of::<libc::$ident>() => fmt::<usize>()
46            );
47            compile_assert!(
48                std::mem::align_of::<sys::$ident>() == std::mem::align_of::<libc::$ident>(),
49                "libc/esp-idf-sys type mismatch for `", stringify!($ident), "` alignment: esp-idf=",
50                std::mem::align_of::<sys::$ident>() => fmt::<usize>(),
51                " libc=", std::mem::align_of::<libc::$ident>() => fmt::<usize>()
52            );
53        };
54    };
55}
56
57macro_rules! check_types_manually {
58    ($name:literal, $size1:expr, $size2:expr, $align1:expr, $align2:expr) => {
59        const _: () = {
60            compile_assert!(
61                $size1 == $size2,
62                "libc/esp-idf-sys type mismatch for `", $name, "` size: lhs=",
63                $size1 => fmt::<usize>(),
64                " rhs=", $size2 => fmt::<usize>()
65            );
66            compile_assert!(
67                $align1 == $align2,
68                "libc/esp-idf-sys type mismatch for `", $name, "` alignment: lhs=",
69                $align1 => fmt::<usize>(),
70                " rhs=", $align2 => fmt::<usize>()
71            );
72        };
73    };
74}

76// Most of these tests can be generated with from rust-lang/libc tree, with:
77// sed -n 's/.*pub type \(.*\) = .*/check_types!(\1);/p' mod.rs | uniq
78// sed -n 's/.*pub struct \(.*\) {.*/check_types!(\1);/p' mod.rs | uniq
79// sed -n 's/.*pub const \(.*\):.*/check_constants!(\1);/p' mod.rs | uniq
80// You should manually check for `cfg` expressions in the source, though.
81// Then you should then manually remove any items that don't have a binding in `esp_idf_sys`.
82// And finally you should remove duplicates.
